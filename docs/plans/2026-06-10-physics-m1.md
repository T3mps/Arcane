**Documented narrowings (M1, spec-consistent):**
- **No `Fixture.lua` yet:** M1 bodies carry exactly one shape + filter + `isSensor` inline. The multi-fixture record splits out when something needs two shapes on one body (M3 era). Accessors are named so the split is non-breaking.
- **Slide is sub-stepped discrete depenetration, not true swept:** ≤8px samples + two-pass push-out gives identical behavior at our speeds; conservative-advancement sweep arrives with GJK/CCD in M2 (where the spec puts it).
- **Static off-grid bodies use a plain list** scanned per mover (M1 scale: a handful of props). `Broadphase`-interface generalization + SAP/AABBTree land in M4 per spec; the query call-sites are already funneled through `PhysicsWorld:_staticCandidates` so the swap is local.
- **P1 invariants inherited (do not regress):** bodies must never overlap solid cell footprints (the P1 "clipping between blocks" lesson — the capsule keeps the whole body out, replacing the interim ±14px feet segment), and large dt must not tunnel (sub-stepping).

**Coordinate/units conventions (used everywhere):** y-down screen space; angles unused (no rotation in M1 — `fixedRotation` world). Normals returned by narrowphase point **from the static/other shape toward the queried shape** (the push-out direction). Polygons are convex, ≤8 verts, flat array `{x1,y1,x2,y2,...}`, any consistent winding (normal orientation is resolved against the centroid, not the winding).

```lua
-- physics/shapes.lua
-- Collider definitions: pure geometry, immutable, shareable across bodies
-- (spec §Core model). No rotation in M1 (fixedRotation world): capsules are
-- horizontal segments (the iso ground-contact convention from P1), polygons
-- are baked in local space.
local shapes = {}

local MAX_POLY_VERTS = 8   -- spec resolved decision #2: single constant

function shapes.circle(r)
    return { kind = "circle", r = r }
end

--- Horizontal capsule: segment (-halfLen,0)-(+halfLen,0) inflated by r.
function shapes.capsule(halfLen, r)
    return { kind = "capsule", halfLen = halfLen, r = r }
end

--- Axis-aligned box by half-extents.
function shapes.aabb(hw, hh)
    return { kind = "aabb", hw = hw, hh = hh }
end

--- Convex polygon, flat vert array {x1,y1,...} in local space, <= 8 verts.
function shapes.polygon(verts)
    assert(#verts >= 6 and #verts % 2 == 0 and #verts <= MAX_POLY_VERTS * 2,
        "polygon: need 3..8 (x,y) verts")
    return { kind = "polygon", verts = verts }
end

--- World AABB of a shape at position (x, y): x0, y0, x1, y1.
function shapes.aabbOf(s, x, y)
    if s.kind == "circle" then
        return x - s.r, y - s.r, x + s.r, y + s.r
    elseif s.kind == "capsule" then
        return x - s.halfLen - s.r, y - s.r, x + s.halfLen + s.r, y + s.r
    elseif s.kind == "aabb" then
        return x - s.hw, y - s.hh, x + s.hw, y + s.hh
    else
        local v = s.verts
        local x0, y0, x1, y1 = math.huge, math.huge, -math.huge, -math.huge
        for i = 1, #v, 2 do
            local vx, vy = v[i] + x, v[i + 1] + y
            if vx < x0 then x0 = vx end; if vx > x1 then x1 = vx end
            if vy < y0 then y0 = vy end; if vy > y1 then y1 = vy end
        end
        return x0, y0, x1, y1
    end
end

return shapes
```

```lua
--- Continuous (fractional) cell coords — the physics/raycast lattice space,
--- where every cell is the unit square centered on its integer coords.
function IsoProjection.toCellF(sx, sy)
    local fx = (sx - HALF_W) / HALF_W
    local fy = (sy - HALF_H) / HALF_H
    return (fx + fy) / 2, (fy - fx) / 2
end

--- Inverse of toCellF.
function IsoProjection.fromCellF(cx, cy)
    return (cx - cy) * HALF_W + HALF_W, (cx + cy) * HALF_H + HALF_H
end
```

```lua
-- physics/Geometry.lua
-- Pure 2D collision kernel (y-down screen space, no rotation). Normals point
-- FROM the second/static shape TOWARD the first/queried shape — i.e. the
-- direction that pushes the queried shape out. Polygons: flat vert arrays,
-- convex, any winding (outward normals resolved against the centroid).
local Geometry = {}

local sqrt, huge = math.sqrt, math.huge

function Geometry.closestOnSegment(px, py, ax, ay, bx, by)
    local abx, aby = bx - ax, by - ay
    local d = abx * abx + aby * aby
    local t = d > 0 and ((px - ax) * abx + (py - ay) * aby) / d or 0
    if t < 0 then t = 0 elseif t > 1 then t = 1 end
    return ax + abx * t, ay + aby * t, t
end

--- Closest points between segments P(p1->p2) and Q(q1->q2): px,py,qx,qy.
function Geometry.closestSegSeg(p1x, p1y, p2x, p2y, q1x, q1y, q2x, q2y)
    -- Sample candidate pairs: each endpoint vs the other segment, plus the
    -- crossing case. For non-degenerate use this is exact: the minimum
    -- distance between two segments is attained at an endpoint of one of
    -- them unless they intersect.
    local bestD, bpx, bpy, bqx, bqy = huge, 0, 0, 0, 0
    local function consider(ax, ay, bx, by)
        local dx, dy = ax - bx, ay - by
        local d = dx * dx + dy * dy
        if d < bestD then bestD, bpx, bpy, bqx, bqy = d, ax, ay, bx, by end
    end
    local cx, cy = Geometry.closestOnSegment(p1x, p1y, q1x, q1y, q2x, q2y)
    consider(p1x, p1y, cx, cy)
    cx, cy = Geometry.closestOnSegment(p2x, p2y, q1x, q1y, q2x, q2y)
    consider(p2x, p2y, cx, cy)
    cx, cy = Geometry.closestOnSegment(q1x, q1y, p1x, p1y, p2x, p2y)
    consider(cx, cy, q1x, q1y)
    cx, cy = Geometry.closestOnSegment(q2x, q2y, p1x, p1y, p2x, p2y)
    consider(cx, cy, q2x, q2y)
    -- Proper crossing: endpoints alone miss it. Solve the 2x2 system.
    local rx, ry = p2x - p1x, p2y - p1y
    local sx, sy = q2x - q1x, q2y - q1y
    local denom = rx * sy - ry * sx
    if math.abs(denom) > 1e-12 then
        local t = ((q1x - p1x) * sy - (q1y - p1y) * sx) / denom
        local u = ((q1x - p1x) * ry - (q1y - p1y) * rx) / denom
        if t >= 0 and t <= 1 and u >= 0 and u <= 1 then
            local ix, iy = p1x + rx * t, p1y + ry * t
            return ix, iy, ix, iy
        end
    end
    return bpx, bpy, bqx, bqy
end

function Geometry.pointInPoly(px, py, verts)
    local n = #verts
    local sign = 0
    for i = 1, n, 2 do
        local j = (i + 1 < n) and i + 2 or 1
        local ex, ey = verts[j] - verts[i], verts[j + 1] - verts[i + 1]
        local cross = ex * (py - verts[i + 1]) - ey * (px - verts[i])
        if cross ~= 0 then
            local s = cross > 0 and 1 or -1
            if sign == 0 then sign = s elseif s ~= sign then return false end
        end
    end
    return true
end

local function polyCentroid(verts)
    local cx, cy, n = 0, 0, #verts / 2
    for i = 1, #verts, 2 do cx = cx + verts[i]; cy = cy + verts[i + 1] end
    return cx / n, cy / n
end

-- Closest point on a polygon's BOUNDARY to p; also the outward normal of the
-- supporting edge (resolved against the centroid).
local function closestOnPolyBoundary(px, py, verts)
    local n = #verts
    local bestD, bx, by, bnx, bny = huge, 0, 0, 0, 0
    local ccx, ccy = polyCentroid(verts)
    for i = 1, n, 2 do
        local j = (i + 1 < n) and i + 2 or 1
        local ax, ay, qx, qy = verts[i], verts[i + 1], verts[j], verts[j + 1]
        local cx2, cy2 = Geometry.closestOnSegment(px, py, ax, ay, qx, qy)
        local dx, dy = px - cx2, py - cy2
        local d = dx * dx + dy * dy
        if d < bestD then
            bestD, bx, by = d, cx2, cy2
            -- outward edge normal
            local ex, ey = qx - ax, qy - ay
            local nx2, ny2 = ey, -ex
            local len = sqrt(nx2 * nx2 + ny2 * ny2)
            if len > 0 then nx2, ny2 = nx2 / len, ny2 / len end
            local mx, my = (ax + qx) / 2 - ccx, (ay + qy) / 2 - ccy
            if nx2 * mx + ny2 * my < 0 then nx2, ny2 = -nx2, -ny2 end
            bnx, bny = nx2, ny2
        end
    end
    return bx, by, sqrt(bestD), bnx, bny
end

--- Circle (cx,cy,r) vs convex poly. Returns hit, nx, ny, depth.
function Geometry.circlePoly(cx, cy, r, verts)
    local bx, by, dist, enx, eny = closestOnPolyBoundary(cx, cy, verts)
    if Geometry.pointInPoly(cx, cy, verts) then
        -- contained: push out along the nearest edge's outward normal
        return true, enx, eny, r + dist
    end
    if dist >= r then return false end
    if dist > 1e-9 then
        return true, (cx - bx) / dist, (cy - by) / dist, r - dist
    end
    return true, enx, eny, r   -- degenerate: center on boundary
end

--- Capsule (segment a-b, radius r) vs convex poly. Returns hit, nx, ny, depth.
function Geometry.capsulePoly(ax, ay, bx, by, r, verts)
    -- Containment (either endpoint inside): treat as a deep circle at the
    -- deeper endpoint. Substepped movement keeps real penetrations shallow.
    if Geometry.pointInPoly(ax, ay, verts) or Geometry.pointInPoly(bx, by, verts) then
        local inx = Geometry.pointInPoly(ax, ay, verts) and ax or bx
        local iny = inx == ax and ay or by
        return Geometry.circlePoly(inx, iny, r, verts)
    end
    -- Shallow case: closest pair between the segment and the boundary edges.
    local n = #verts
    local bestD, snx, sny = huge, 0, 0
    for i = 1, n, 2 do
        local j = (i + 1 < n) and i + 2 or 1
        local px, py, qx, qy = Geometry.closestSegSeg(
            ax, ay, bx, by, verts[i], verts[i + 1], verts[j], verts[j + 1])
        local dx, dy = px - qx, py - qy
        local d = sqrt(dx * dx + dy * dy)
        if d < bestD then
            bestD = d
            if d > 1e-9 then snx, sny = dx / d, dy / d end
        end
    end
    if bestD >= r then return false end
    if snx == 0 and sny == 0 then
        -- segment touches boundary exactly: borrow the edge normal
        local _, _, _, enx, eny = closestOnPolyBoundary(ax, ay, verts)
        snx, sny = enx, eny
    end
    return true, snx, sny, r - bestD
end

--- Circle vs circle (A queried against B). Returns hit, nx, ny, depth.
function Geometry.circleCircle(ax, ay, ar, bx, by, br)
    local dx, dy = ax - bx, ay - by
    local d = sqrt(dx * dx + dy * dy)
    local rsum = ar + br
    if d >= rsum then return false end
    if d > 1e-9 then return true, dx / d, dy / d, rsum - d end
    return true, 1, 0, rsum
end

--- SAT convex poly vs convex poly. Returns hit, nx, ny, depth (push A out of B).
function Geometry.polyPoly(va, vb)
    local bestDepth, bnx, bny = huge, 0, 0
    local function axes(verts, other)
        local n = #verts
        for i = 1, n, 2 do
            local j = (i + 1 < n) and i + 2 or 1
            local nx, ny = verts[j + 1] - verts[i + 1], verts[i] - verts[j]
            local len = sqrt(nx * nx + ny * ny)
            if len > 1e-12 then
                nx, ny = nx / len, ny / len
                local minA, maxA, minB, maxB = huge, -huge, huge, -huge
                for k = 1, #verts, 2 do
                    local p = verts[k] * nx + verts[k + 1] * ny
                    if p < minA then minA = p end; if p > maxA then maxA = p end
                end
                for k = 1, #other, 2 do
                    local p = other[k] * nx + other[k + 1] * ny
                    if p < minB then minB = p end; if p > maxB then maxB = p end
                end
                local overlap = math.min(maxA, maxB) - math.max(minA, minB)
                if overlap <= 0 then return false end
                if overlap < bestDepth then
                    bestDepth = overlap
                    -- orient from B toward A
                    local ca, cb = (minA + maxA) / 2, (minB + maxB) / 2
                    if ca >= cb then bnx, bny = nx, ny else bnx, bny = -nx, -ny end
                end
            end
        end
        return true
    end
    if not axes(va, vb) then return false end
    if not axes(vb, va) then return false end
    return true, bnx, bny, bestDepth
end

function Geometry.aabbOverlap(ax0, ay0, ax1, ay1, bx0, by0, bx1, by1)
    return ax0 <= bx1 and bx0 <= ax1 and ay0 <= by1 and by0 <= ay1
end

return Geometry
```

```lua
-- Static broadphase that stores NOTHING for walls: it answers queries
-- straight from Map cell flags (spec §Broadphase — single-source-of-truth:
-- change a cell flag, physics follows instantly). Solid cells are diamonds
-- in screen space; a row-run of edge-adjacent diamonds unions exactly into a
-- PARALLELOGRAM, which is this grid's "virtual fixture" (the spec's greedy
-- rectangle merge expressed in iso space — kills internal-edge catching
-- along rows). Verts are written into caller-provided pooled rows:
--   out[i] = { x1,y1, x2,y2, x3,y3, x4,y4 }  (top, right, bottom, left)

local HALF_H = Iso.TILE_H / 2

end

--- Fill `out` with merged solid spans intersecting the world-space AABB.
--- Returns the span count. Rows of `out` beyond the count are stale; rows
--- are reused across calls (zero steady-state allocation after warmup).
function TileGrid:query(x0, y0, x1, y1, out)
    local map = self.map
    local minFx, minFy = math.huge, math.huge
    local maxFx, maxFy = -math.huge, -math.huge
    local function corner(sx, sy)
        local fx, fy = Iso.toCellF(sx, sy)
        if fx < minFx then minFx = fx end
        if fx > maxFx then maxFx = fx end
        if fy < minFy then minFy = fy end
        if fy > maxFy then maxFy = fy end
    end
    corner(x0, y0); corner(x1, y0); corner(x0, y1); corner(x1, y1)
    local a = math.max(1, math.floor(minFx + 0.5) - PAD)
    local b = math.max(1, math.floor(minFy + 0.5) - PAD)
    local c = math.min(map.w, math.floor(maxFx + 0.5) + PAD)
    local d = math.min(map.h, math.floor(maxFy + 0.5) + PAD)

    local n = 0
    for cy = b, d do
        local cx = a
        while cx <= c do
            if not map:isWalkable(cx, cy) then
                local rx0 = cx
                while cx + 1 <= c and not map:isWalkable(cx + 1, cy) do cx = cx + 1 end
                n = n + 1
                local row = out[n]
                if not row then row = {}; out[n] = row end
                local sx0, sy0 = Iso.toScreen(rx0, cy)
                local sx1, sy1 = Iso.toScreen(cx, cy)
                row[1], row[2] = sx0 + HALF_W,      sy0                -- top of first
                row[3], row[4] = sx1 + Iso.TILE_W,  sy1 + HALF_H       -- right of last
                row[5], row[6] = sx1 + HALF_W,      sy1 + Iso.TILE_H   -- bottom of last
                row[7], row[8] = sx0,               sy0 + HALF_H       -- left of first
            end
            cx = cx + 1
        end
    end
    return n
end

return TileGrid
```

```lua
-- physics/SpatialHash.lua
-- Uniform-grid broadphase for movers (spec: keyed ~2x max mover extent).
-- Entries are integer ids (body indices). Buckets are arrays; an entry
-- remembers its bucket-key range so update/remove are exact.
local SpatialHash = {}
SpatialHash.__index = SpatialHash

local floor = math.floor

function SpatialHash.new(cellSize)
    return setmetatable({
        cs      = cellSize or 64,
        buckets = {},   -- key -> array of ids
        ranges  = {},   -- id -> {kx0, ky0, kx1, ky1}
        _seen   = {},   -- pair dedupe scratch (generation-stamped)
        _gen    = 0,
    }, SpatialHash)
end

local function bkey(kx, ky) return kx * 0x40000 + ky end   -- fine for world scale

local function removeFromBucket(bucket, id)
    for i = #bucket, 1, -1 do
        if bucket[i] == id then bucket[i] = bucket[#bucket]; bucket[#bucket] = nil; return end
    end
end

function SpatialHash:remove(id)
    local r = self.ranges[id]
    if not r then return end
    for kx = r[1], r[3] do for ky = r[2], r[4] do
        local b = self.buckets[bkey(kx, ky)]
        if b then removeFromBucket(b, id) end
    end end
    self.ranges[id] = nil
end

function SpatialHash:update(id, x0, y0, x1, y1)
    local cs = self.cs
    local kx0, ky0 = floor(x0 / cs), floor(y0 / cs)
    local kx1, ky1 = floor(x1 / cs), floor(y1 / cs)
    local r = self.ranges[id]
    if r and r[1] == kx0 and r[2] == ky0 and r[3] == kx1 and r[4] == ky1 then return end
    self:remove(id)
    for kx = kx0, kx1 do for ky = ky0, ky1 do
        local k = bkey(kx, ky)
        local b = self.buckets[k]
        if not b then b = {}; self.buckets[k] = b end
        b[#b + 1] = id
    end end
    self.ranges[id] = r or {}
    r = self.ranges[id]
    r[1], r[2], r[3], r[4] = kx0, ky0, kx1, ky1
end

--- Ids whose bucket range intersects the AABB. Returns count; fills out[].
function SpatialHash:queryAABB(x0, y0, x1, y1, out)
    local cs = self.cs
    self._gen = self._gen + 1
    local gen, seen = self._gen, self._seen
    local n = 0
    for kx = floor(x0 / cs), floor(x1 / cs) do
        for ky = floor(y0 / cs), floor(y1 / cs) do
            local b = self.buckets[bkey(kx, ky)]
            if b then
                for i = 1, #b do
                    local id = b[i]
                    if seen[id] ~= gen then
                        seen[id] = gen
                        n = n + 1; out[n] = id
                    end
                end
            end
        end
    end
    return n
end

--- Candidate pairs (shared bucket), deduped, a<b, written flat into out.
--- Returns the PAIR count (out holds 2*count entries).
function SpatialHash:pairs(out)
    self._gen = self._gen + 1
    local gen, seen = self._gen, self._seen
    local n = 0
    for _, b in pairs(self.buckets) do
        for i = 1, #b do
            for j = i + 1, #b do
                local a, c = b[i], b[j]
                if a > c then a, c = c, a end
                local pk = a * 0x10000 + c
                if seen[pk] ~= gen then
                    seen[pk] = gen
                    out[n * 2 + 1], out[n * 2 + 2] = a, c
                    n = n + 1
                end
            end
        end
    end
    return n
end

return SpatialHash
```

```lua
-- physics/ContactManager.lua
-- Persistent contact pairs over the mover broadphase: begin/stay/end events,
-- queued during step and delivered AFTER it (spec: no user code mutates
-- mid-solve). Pairs: mover-mover (SpatialHash) and mover-staticBody (list).
-- Mover-vs-TILE contacts deliberately do not flow here: the character
-- controller owns its statics response, and cells are not bodies.
-- Event gating (per-body + world, drop-don't-queue, level-triggered re-arm)
-- lands in Task 7 on top of the `begun` flag this task introduces.
local shapes   = require "physics.shapes"
local Geometry = require "physics.Geometry"

local ContactManager = {}
ContactManager.__index = ContactManager

function ContactManager.new()
    return setmetatable({
        pairs    = {},   -- key -> { a, b, touching, begun }
        listener = nil,
        _queue   = {},   -- pooled event rows
        _qn      = 0,
        _stamp   = 0,
    }, ContactManager)
end

local function pairKey(a, b)
    if a > b then a, b = b, a end
    return a * 0x10000 + b
end

-- AABB shape -> polygon verts (scratch) for the unified poly tests
local _polyScratch = {}
local function shapePoly(s, x, y)
    if s.kind == "aabb" then
        local v = _polyScratch
        v[1], v[2] = x - s.hw, y - s.hh
        v[3], v[4] = x + s.hw, y - s.hh
        v[5], v[6] = x + s.hw, y + s.hh
        v[7], v[8] = x - s.hw, y + s.hh
        return v
    end
    return nil
end

--- Overlap test for any M1 shape pair at world positions.
function ContactManager.shapesOverlap(sa, ax, ay, sb, bx, by)
    -- normalize: treat circle as zero-length capsule
    local function asCapsule(s, x, y)
        if s.kind == "circle" then return x, y, x, y, s.r end
        if s.kind == "capsule" then return x - s.halfLen, y, x + s.halfLen, y, s.r end
        return nil
    end
    local a1x, a1y, a2x, a2y, ar = asCapsule(sa, ax, ay)
    local b1x, b1y, b2x, b2y, br = asCapsule(sb, bx, by)
    if ar and br then
        local px, py, qx, qy = Geometry.closestSegSeg(a1x, a1y, a2x, a2y, b1x, b1y, b2x, b2y)
        local dx, dy = px - qx, py - qy
        return (dx * dx + dy * dy) < (ar + br) * (ar + br)
    end
    local pa = shapePoly(sa, ax, ay)
    if pa and br then return Geometry.capsulePoly(b1x, b1y, b2x, b2y, br, pa) end
    local pb = shapePoly(sb, bx, by)
    if pb and ar then return Geometry.capsulePoly(a1x, a1y, a2x, a2y, ar, pb) end
    if pa and pb then
        -- both AABBs: cheap test (pa scratch would alias; use direct AABB)
        local ax0, ay0, ax1, ay1 = shapes.aabbOf(sa, ax, ay)
        local bx0, by0, bx1, by1 = shapes.aabbOf(sb, bx, by)
        return Geometry.aabbOverlap(ax0, ay0, ax1, ay1, bx0, by0, bx1, by1)
    end
    if sa.kind == "polygon" and pb then return Geometry.polyPoly(sa.verts, pb) end
    if sb.kind == "polygon" and pa then return Geometry.polyPoly(pa, sb.verts) end
    if sa.kind == "polygon" and sb.kind == "polygon" then
        return Geometry.polyPoly(sa.verts, sb.verts)   -- local==world (no offset polys in M1 runtime)
    end
    return false
end

function ContactManager:_emit(w, type_, a, b)
    -- Gating (Task 7 finalizes; the begun bookkeeping lives in step below)
    if not w.eventsEnabled then return end
    if w.evtOn[a] == 0 or w.evtOn[b] == 0 then return end
    if not self.listener then return end
    self._qn = self._qn + 1
    local ev = self._queue[self._qn]
    if not ev then ev = {}; self._queue[self._qn] = ev end
    ev.type   = type_
    ev.a      = w.handles[a]
    ev.b      = w.handles[b]
    ev.sensor = (w.sensor[a] == 1) or (w.sensor[b] == 1)
end

function ContactManager:step(w)
    self._stamp = self._stamp + 1
    local stamp = self._stamp
    self._qn = 0

    -- candidates: mover-mover + mover-staticBody
    local mm = w._pairs
    local np = w.moverHash:pairs(mm)
    local function touch(a, b)
        if w.alive[a] == 0 or w.alive[b] == 0 then return end
        if ContactManager.shapesOverlap(w.shape[a], w.posX[a], w.posY[a],
                                        w.shape[b], w.posX[b], w.posY[b]) then
            local k = pairKey(a, b)
            local p = self.pairs[k]
            if not p then p = { a = math.min(a,b), b = math.max(a,b), begun = false }; self.pairs[k] = p end
            p.stamp = stamp
            if not p.begun then
                p.begun = true
                self:_emit(w, "begin", p.a, p.b)
            else
                self:_emit(w, "stay", p.a, p.b)
            end
        end
    end
    for i = 1, np * 2, 2 do touch(mm[i], mm[i + 1]) end
    -- movers vs static bodies (index-ordered for determinism)
    for i = 0, w.count - 1 do
        if w.alive[i] == 1 and w.btype[i] == w.KINEMATIC then
            for s = 1, #w.staticList do touch(i, w.staticList[s]) end
        end
    end

    -- pairs not touched this step have separated
    for k, p in pairs(self.pairs) do
        if p.stamp ~= stamp then
            if p.begun then self:_emit(w, "end", p.a, p.b) end
            self.pairs[k] = nil
        end
    end

    -- flush AFTER all state is settled
    if self.listener then
        for i = 1, self._qn do self.listener(self._queue[i]) end
    end
end

function ContactManager:dropBody(idx)
    for k, p in pairs(self.pairs) do
        if p.a == idx or p.b == idx then self.pairs[k] = nil end
    end
end

return ContactManager
```

```lua
local Iso = require "world.IsoProjection"   -- top of file

--- Raycast against solid cells (DDA over the fractional-cell lattice, where
--- cells are unit squares centered on integers) and body shapes (AABB-slab /
--- segment-vs-circle). Returns nil or { x, y, t, cellX, cellY | body }.
--- M2 extends this with TALL-only filtering for Combat Sphere LOS.
function PhysicsWorld:raycast(x0, y0, x1, y1)
    if not x1 then return nil end
    local best   -- { t, x, y, cellX, cellY, body }

    -- cells via DDA in fractional cell space
    if self.map then
        local fx0, fy0 = Iso.toCellF(x0, y0)
        local fx1, fy1 = Iso.toCellF(x1, y1)
        local dx, dy = fx1 - fx0, fy1 - fy0
        local cx, cy = math.floor(fx0 + 0.5), math.floor(fy0 + 0.5)
        local endCx, endCy = math.floor(fx1 + 0.5), math.floor(fy1 + 0.5)
        local stepX = dx > 0 and 1 or -1
        local stepY = dy > 0 and 1 or -1
        -- param t to the next half-integer boundary on each axis
        local function boundaryT(f0, d, cell, step)
            if d == 0 then return math.huge end
            local boundary = cell + step * 0.5
            return (boundary - f0) / d
        end
        local tMaxX = boundaryT(fx0, dx, cx, stepX)
        local tMaxY = boundaryT(fy0, dy, cy, stepY)
        local tDeltaX = dx ~= 0 and math.abs(1 / dx) or math.huge
        local tDeltaY = dy ~= 0 and math.abs(1 / dy) or math.huge
        local t = 0
        for _ = 1, (self.map.w + self.map.h) * 2 do   -- bounded expansion
            if not self.map:isWalkable(cx, cy) then
                local hx = x0 + (x1 - x0) * t
                local hy = y0 + (y1 - y0) * t
                best = { t = t, x = hx, y = hy, cellX = cx, cellY = cy }
                break
            end
            if cx == endCx and cy == endCy then break end
            if tMaxX < tMaxY then
                t = tMaxX; tMaxX = tMaxX + tDeltaX; cx = cx + stepX
            else
                t = tMaxY; tMaxY = tMaxY + tDeltaY; cy = cy + stepY
            end
            if t > 1 then break end
        end
    end

    -- bodies: segment vs shape AABB (slab test; M1 precision is AABB-level,
    -- exact shape tests arrive with GJK in M2)
    for i = 0, self.count - 1 do
        if self.alive[i] == 1 then
            local a0, b0, a1, b1 = require("physics.shapes").aabbOf(self.shape[i], self.posX[i], self.posY[i])
            local tmin, tmax = 0, 1
            local dx, dy = x1 - x0, y1 - y0
            local ok = true
            local function slab(o, d, lo, hi)
                if d == 0 then if o < lo or o > hi then ok = false end return end
                local ta, tb = (lo - o) / d, (hi - o) / d
                if ta > tb then ta, tb = tb, ta end
                if ta > tmin then tmin = ta end
                if tb < tmax then tmax = tb end
            end
            slab(x0, dx, a0, a1); slab(y0, dy, b0, b1)
            if ok and tmin <= tmax and tmin <= 1 and tmax >= 0 then
                local t = math.max(0, tmin)
                if not best or t < best.t then
                    best = { t = t, x = x0 + dx * t, y = y0 + dy * t, body = self.handles[i] }
                end
            end
        end
    end
    return best
end
```

```lua
-- physics/CharacterController.lua
-- Game-facing character motion (spec §Character control).
--   slideMove(dx, dy): WASD — substepped capsule depenetration with two-pass
--     slide vs TileGrid spans + static bodies. Discrete substeps (<= 8px)
--     stand in for the true swept test until GJK/CCD (M2); combined with
--     depenetration this preserves both P1 invariants: never buried in a
--     solid footprint, never tunneling on a dt spike.
--   followVelocity(vx, vy): click mode — plain kinematic velocity; physics
--     integrates it and only reports overlaps (paths are cell-safe by
--     construction; the follower is never deflected).
local shapes   = require "physics.shapes"
local Geometry = require "physics.Geometry"

local CharacterController = {}
CharacterController.__index = CharacterController

local MAX_SUBSTEP = 8
local MAX_PASSES  = 3
local SKIN        = 0.05   -- resolve to a hair outside the surface

function CharacterController.new(world, body)
    return setmetatable({
        world = world,
        body  = body,
        _spans = {}, _statics = {},
        _polyScratch = {},
    }, CharacterController)
end

local function aabbPoly(scratch, x, y, hw, hh)
    scratch[1], scratch[2] = x - hw, y - hh
    scratch[3], scratch[4] = x + hw, y - hh
    scratch[5], scratch[6] = x + hw, y + hh
    scratch[7], scratch[8] = x - hw, y + hh
    return scratch
end

--- Push the capsule at (x,y) out of all nearby statics. Returns resolved x,y.
function CharacterController:_depenetrate(x, y)
    local w, s = self.world, self.body:shape()
    for _ = 1, MAX_PASSES do
        local x0, y0, x1, y1 = shapes.aabbOf(s, x, y)
        local nSpans, nStat = w:_staticCandidates(x0, y0, x1, y1, self._spans, self._statics)
        local hit, bnx, bny, bdepth = false, 0, 0, 0
        for i = 1, nSpans do
            local h, nx, ny, depth = Geometry.capsulePoly(
                x - s.halfLen, y, x + s.halfLen, y, s.r, self._spans[i])
            if h and depth > bdepth then hit, bnx, bny, bdepth = true, nx, ny, depth end
        end
        for i = 1, nStat do
            local idx = self._statics[i]
            if w.sensor[idx] == 0 then
                local os = w.shape[idx]
                local poly
                if os.kind == "aabb" then
                    poly = aabbPoly(self._polyScratch, w.posX[idx], w.posY[idx], os.hw, os.hh)
                elseif os.kind == "polygon" then
                    poly = os.verts
                end
                if poly then
                    local h, nx, ny, depth = Geometry.capsulePoly(
                        x - s.halfLen, y, x + s.halfLen, y, s.r, poly)
                    if h and depth > bdepth then hit, bnx, bny, bdepth = true, nx, ny, depth end
                end
            end
        end
        if not hit then return x, y end
        x = x + bnx * (bdepth + SKIN)
        y = y + bny * (bdepth + SKIN)
    end
    return x, y
end

--- Move by (dx, dy) with slide. Returns the ACTUAL displacement.
function CharacterController:slideMove(dx, dy)
    local body = self.body
    local sx, sy = body:getPosition()
    local x, y = sx, sy
    local dist = math.sqrt(dx * dx + dy * dy)
    if dist < 1e-9 then return 0, 0 end
    local ux, uy = dx / dist, dy / dist
    local remaining = dist
    while remaining > 0 do
        local stepLen = math.min(remaining, MAX_SUBSTEP)
        remaining = remaining - stepLen
        x, y = self:_depenetrate(x + ux * stepLen, y + uy * stepLen)
    end
    -- write back through the body (prev untouched: the WORLD step manages
    -- prev; callers run slideMove between steps via direct position set)
    local w, i = body.world, body.idx
    w.posX[i], w.posY[i] = x, y
    local x0, y0, x1, y1 = shapes.aabbOf(body:shape(), x, y)
    w.moverHash:update(i, x0, y0, x1, y1)
    return x - sx, y - sy
end

--- Click-to-move: hand the velocity to the kinematic body; world:step
--- integrates it without deflection.
function CharacterController:followVelocity(vx, vy)
    self.body:setVelocity(vx, vy)
end

return CharacterController
```

**Integration contract (consumed by the companion plan — ORDER MATTERS):** the caller runs `world:step(dt)` FIRST, then `slideMove` (WASD) / `followVelocity` (click), every fixedUpdate. Rationale: `step` snapshots `prev` at its top; with the slide applied AFTER the step, `prev` holds the pre-slide position and the render boundary's prev→pos lerp spans exactly this tick's slide (running the slide before the step would make `prev == pos` and kill WASD interpolation). In WASD mode the body's velocity stays 0 so step's integration never double-moves; in click mode the velocity set this tick is integrated by the NEXT step (one tick of latency at constant speed — invisible). Consequence: contact events computed in `step` see positions from the previous controller tick — one tick of event latency, accepted for M1.

```lua
-- physics/PhysicsDebug.lua
-- The "physics" DebugDraw channel: body shapes, mover AABBs, TileGrid spans
-- near movers, and live contact pairs. Registered by the game integration
-- (companion plan) via:
--   DebugDraw.channel("physics", function(map, camera)
--       PhysicsDebug.draw(state.physicsWorld, camera)
--   end, "debugDrawPhysics")
local shapes = require "physics.shapes"

local PhysicsDebug = {}

local COL_KINEMATIC = { 0.2, 1.0, 0.4, 1.0 }
local COL_STATIC    = { 0.4, 0.7, 1.0, 1.0 }
local COL_SENSOR    = { 1.0, 0.9, 0.2, 0.9 }
local COL_SPAN      = { 1.0, 0.3, 0.3, 0.6 }
local COL_CONTACT   = { 1.0, 0.2, 1.0, 1.0 }

local _spans, _statics = {}, {}

function PhysicsDebug.draw(world, camera)
    if not world then return end
    local lg = love.graphics
    local prevWidth = lg.getLineWidth()
    lg.setLineWidth(1 / camera:renderZoom())

    for i = 0, world.count - 1 do
        if world.alive[i] == 1 then
            local s, x, y = world.shape[i], world.posX[i], world.posY[i]
            lg.setColor(world.sensor[i] == 1 and COL_SENSOR
                or world.btype[i] == world.KINEMATIC and COL_KINEMATIC or COL_STATIC)
            if s.kind == "circle" then
                lg.circle("line", x, y, s.r)
            elseif s.kind == "capsule" then
                lg.circle("line", x - s.halfLen, y, s.r)
                lg.circle("line", x + s.halfLen, y, s.r)
                lg.line(x - s.halfLen, y - s.r, x + s.halfLen, y - s.r)
                lg.line(x - s.halfLen, y + s.r, x + s.halfLen, y + s.r)
            elseif s.kind == "aabb" then
                lg.rectangle("line", x - s.hw, y - s.hh, s.hw * 2, s.hh * 2)
            elseif s.kind == "polygon" then
                lg.push(); lg.translate(x, y); lg.polygon("line", s.verts); lg.pop()
            end
            -- TileGrid spans near each mover
            if world.btype[i] == world.KINEMATIC and world.tileGrid then
                local x0, y0, x1, y1 = shapes.aabbOf(s, x, y)
                local n = world:_staticCandidates(x0 - 32, y0 - 32, x1 + 32, y1 + 32, _spans, _statics)
                lg.setColor(COL_SPAN)
                for k = 1, n do lg.polygon("line", _spans[k]) end
            end
        end
    end

    -- live contact pairs
    if world.contacts then
        lg.setColor(COL_CONTACT)
        for _, p in pairs(world.contacts.pairs) do
            if p.begun then
                lg.line(world.posX[p.a], world.posY[p.a], world.posX[p.b], world.posY[p.b])
            end
        end
    end

    lg.setLineWidth(prevWidth)
    lg.setColor(1, 1, 1, 1)
end

return PhysicsDebug
```
